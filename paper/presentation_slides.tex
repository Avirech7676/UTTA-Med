\begin{frame}{Research Question and Objectives}
    \begin{block}{Central Research Question}
        \centering\large
        \textit{Can test-time adaptation achieve guaranteed positive transfer in computational pathology without falling into single-class attractor collapse or violating clinical safety bounds?}
    \end{block}

    \vspace{0.3cm}
    \textbf{Three Concrete Research Objectives:}
    \begin{enumerate}
        \item \textbf{Analyze the Failure Mechanism:} Mathematically characterize why confidence and scalar uncertainty gating fail to prevent representation collapse under class imbalance.
        \item \textbf{Formulate Geometry-Preserving Alignment:} Enforce optimal transport with exact marginal prior constraints against certified source prototypes ($\mathbf{Q}^T \mathbf{1}_B = \mathbf{r}$).
        \item \textbf{Engineer Clinical Fail-Closed Safeguards:} Bound parameter drift ($\|\theta_t - \theta_0\|_\infty \le 0.05$), eliminate false silence alarms on acellular stroma, and ensure patient isolation via per-WSI resets.
    \end{enumerate}
\end{frame}

\begin{frame}{Method: Entropic Optimal Transport Precludes Attractor Collapse}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{The Mathematical Formulation:}
            \begin{equation*}
                \min_{\mathbf{Q} \in \mathcal{U}(\mathbf{p}, \mathbf{r})} \langle \mathbf{Q}, \mathbf{M} \rangle - \epsilon H(\mathbf{Q}), \quad \mathbf{M}_{ik} = -\log p(y=k \mid x_i)
            \end{equation*}
            \begin{itemize}
                \item $\mathbf{p} = \frac{1}{B} \mathbf{1}_B$ (uniform patch weight constraint).
                \item $\mathbf{r} = [r_0, r_1]^T$ (certified target class marginal prior).
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Why Attractor Collapse is Impossible:}
            \begin{itemize}
                \item The marginal constraint $\mathbf{Q}^T \mathbf{1}_B = \mathbf{r}$ \textbf{forces} the transportation matrix to assign exactly $r_1 \cdot B$ mass to the tumor class.
                \item Solved via rapid Sinkhorn-Knopp iterations in $< 2.5$\,ms per batch on GPU.

\begin{frame}{Method: Tissue Optical Density Safeguards and PAC Drift Bounds}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Tissue Gating via Beer-Lambert Law}}
                \small
                \textbf{Problem:} Dense mitosis detectors (MIDOG 2021) misinterpret tumor-free acellular stroma as model silence, latching false alarm rollbacks.
                \vspace{0.1cm}\\
                \textbf{Solution:} \textbf{Cellularity-Mitosis Ratio (CMR)}:
                \begin{equation*}
                    \text{OD} = -\log_{10}(\mathbf{I} / \mathbf{I}_0), \quad \mathcal{H} = \mathbf{v}_{\text{hematoxylin}}^T \text{OD}
                \end{equation*}
                Silences adaptation triggers if tissue cellularity $\mathcal{H} < \tau_{\text{cellularity}}$, \textbf{eliminating all 14 false silence alarms}.
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Provable PAC $L_\infty$ Trust Region}}
                \small
                \textbf{Problem:} Parameter drift during continuous adaptation can corrupt decision boundaries over thousands of patches.
                \vspace{0.1cm}\\
                \textbf{Theorem (Drift Projection):}
                Confining updates to $\|\theta_t - \theta_0\|_\infty \le 0.05$ on terminal normalization layers strictly bounds output probability drift:
                \begin{equation*}
                    |\sigma(z_t) - \sigma(z_0)| \le \Delta p_{\max} = \mathbf{0.0460}
                \end{equation*}
                Guarantees fail-closed safety under any corrupt stream.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Ablation Study: Gating Selection and the Taylor Trap Disproof}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Does Gating Mechanism Matter?}}
                \small
                \textbf{Matched-Coverage Control:}
                \begin{itemize}
                    \item Dropping 30\% of patches randomly (\textit{Random-Gated}) yields catastrophic collapse on Seeds 7 and 99 (AUROC $\approx 0.35$).
                    \item \textbf{Insight:} Gains do not come from simply taking fewer gradient steps; \textbf{sample identity is paramount}.
                \end{itemize}
            \end{block}
            \vspace{0.1cm}
            \begin{block}{\textbf{Adaptation Steps ($k$) Sensitivity}}
                \small
                For $k=1$, AUROC = $0.977$; for $k \ge 5$, AUROC plummets to $0.375$. \textbf{Single-step updates ($k=1$) are strictly optimal.}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{\textbf{The Binary Uncertainty Taylor Trap}}
                \small
                \textbf{Common Assumption:} Bayesian MC-Dropout provides orthogonal epistemic information over softmax confidence.
                \vspace{0.1cm}\\
                \textbf{Our Mathematical Disproof:}
                Via Taylor expansion around $\mu_z$:
                \begin{equation*}
                    \text{Var}_{w}[p(y=1 \mid x, w)] \approx p_0^2 (1 - p_0)^2 \sigma_z^2
                \end{equation*}
                \begin{itemize}
                    \item Empirical Spearman correlation: \alert{$\mathbf{\rho = -0.9917}$}.
                    \item MC-Dropout incurs a \alert{$\mathbf{3.55\times}$ latency penalty} with zero incremental sample discrimination.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Honest Limitations and Methodological Boundary Conditions}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{\textbf{1. Multi-Step Drift Sensitivity}}
                \small
                SafeAdapt requires single-step adaptation ($k=1$). Setting $k \ge 5$ triggers representation collapse even with selective gating, as accumulated affine drift overpowers the margin.
            \end{alertblock}
            \vspace{0.2cm}
            \begin{alertblock}{\textbf{2. Class Prior Dependence}}
                \small
                Optimal transport requires a coarse estimate of the target class marginal prior ($\mathbf{r}$). While robust to $\pm 15\%$ estimation errors, radical prior shifts require adaptive estimation.
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{\textbf{3. Architectural Focus}}
                \small
                Evaluated rigorously on ResNet-18 (Camelyon17) and Faster R-CNN R50-FPN (MIDOG 2021). Deeper architectures (e.g., Vision Transformers, ResNet-50) are verified in code but require further multicenter compute runs.
            \end{alertblock}
            \vspace{0.2cm}
            \begin{alertblock}{\textbf{4. Streaming Batch Requirement}}
                \small
                Optimal Transport Sinkhorn alignment operates over mini-batches ($B \ge 32$) to compute empirical margins; single-sample ($B=1$) streaming requires running buffer memory.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}
